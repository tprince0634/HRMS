DELIMITER $$
CREATE PROCEDURE sp_login_user(
    IN p_email VARCHAR(255),
    IN p_password VARCHAR(255)
)
BEGIN
    SELECT
        u.UserId,
        u.Email,
        u.FirstName,
        u.LastName,
        u.RoleId,
        r.RoleName
    FROM `User` u
    JOIN `Role` r
        ON u.RoleId = r.RoleId
    WHERE u.Email = p_email
      AND u.PasswordHash = p_password;
END $$
DELIMITER ;

-- 1. Upload a single file
CREATE PROCEDURE upload_file(
    IN p_fileName LONGTEXT,
    IN p_filePath LONGTEXT,
    IN p_userId INT
)
BEGIN
    INSERT INTO FileUploads (FileName, FilePath, UserId)
    VALUES (p_fileName, p_filePath, p_userId);

    SELECT
        id,
        FileName,
        FilePath,
        UserId
    FROM FileUploads
    WHERE id = LAST_INSERT_ID();
END $$


-- 2. Find file by ID
CREATE PROCEDURE find_file_by_id(
    IN p_id INT
)
BEGIN
    SELECT
        id,
        FileName,
        FilePath,
        UserId
    FROM FileUploads
    WHERE id = p_id;
END $$


-- 3. Find all files uploaded by a user
CREATE PROCEDURE find_files_by_user_id(
    IN p_userId INT
)
BEGIN
    SELECT
        id,
        FileName,
        FilePath,
        UserId
    FROM FileUploads
    WHERE UserId = p_userId
    ORDER BY id DESC;
END $$

-- =========================================================
-- 6. SEARCH TRAINER BY NAME OR EMAIL
-- =========================================================

CREATE PROCEDURE search_trainer_by_name_or_email(
    IN p_name VARCHAR(255),
    IN p_email VARCHAR(255)
)
BEGIN

    SELECT
        TrainerId,
        FirstName,
        LastName,
        Role,
        Email,
        Description,
        Status,
        Phone,
        ProfilePicture
    FROM Trainer
    WHERE
        (
            p_name IS NOT NULL
            AND p_name <> ''
            AND (
                FirstName LIKE CONCAT('%', p_name, '%')
                OR LastName LIKE CONCAT('%', p_name, '%')
            )
        )
        OR
        (
            p_email IS NOT NULL
            AND p_email <> ''
            AND Email LIKE CONCAT('%', p_email, '%')
        )
    ORDER BY TrainerId DESC;

END $$

-- =========================================================
-- 9. GET TRAININGS BY TIME PERIOD
-- DAO: getTrainingsByTime(int page, int pageSize, String period)
--
-- Supported periods:
-- TODAY
-- LAST_7_DAYS
-- LAST_MONTH
-- =========================================================

CREATE PROCEDURE get_trainings_by_time(
    IN p_page INT,
    IN p_pageSize INT,
    IN p_period VARCHAR(50)
)
BEGIN

    DECLARE v_offset INT;

    SET v_offset = (p_page - 1) * p_pageSize;

    IF UPPER(p_period) = 'TODAY' THEN

        SELECT
            TrainingId,
            TrainerId,
            TrainingTypeId,
            UserId,
            TrainingCost,
            Description,
            Status,
            StartDate,
            EndDate,
            CreatedAt,
            CreatedBy,
            ModifiedBy,
            ModifiedAt
        FROM Training
        WHERE StartDate >= CURDATE()
          AND StartDate < CURDATE() + INTERVAL 1 DAY
        ORDER BY StartDate DESC
        LIMIT v_offset, p_pageSize;


    ELSEIF UPPER(p_period) = 'LAST_7_DAYS' THEN

        SELECT
            TrainingId,
            TrainerId,
            TrainingTypeId,
            UserId,
            TrainingCost,
            Description,
            Status,
            StartDate,
            EndDate,
            CreatedAt,
            CreatedBy,
            ModifiedBy,
            ModifiedAt
        FROM Training
        WHERE StartDate >= NOW() - INTERVAL 7 DAY
          AND StartDate <= NOW()
        ORDER BY StartDate DESC
        LIMIT v_offset, p_pageSize;


    ELSEIF UPPER(p_period) = 'LAST_MONTH' THEN

        SELECT
            TrainingId,
            TrainerId,
            TrainingTypeId,
            UserId,
            TrainingCost,
            Description,
            Status,
            StartDate,
            EndDate,
            CreatedAt,
            CreatedBy,
            ModifiedBy,
            ModifiedAt
        FROM Training
        WHERE StartDate >= NOW() - INTERVAL 1 MONTH
          AND StartDate <= NOW()
        ORDER BY StartDate DESC
        LIMIT v_offset, p_pageSize;

    ELSE

        SELECT
            TrainingId,
            TrainerId,
            TrainingTypeId,
            UserId,
            TrainingCost,
            Description,
            Status,
            StartDate,
            EndDate,
            CreatedAt,
            CreatedBy,
            ModifiedBy,
            ModifiedAt
        FROM Training
        ORDER BY StartDate DESC
        LIMIT v_offset, p_pageSize;

    END IF;

END $$
