DELIMITER $$
CREATE PROCEDURE sp_login_user(
    IN p_email VARCHAR(255),
    IN p_password VARCHAR(255)
)
BEGIN
    SELECT
        u.UserId,
        u.Email,
        u.FirstName,
        u.LastName,
        u.RoleId,
        r.RoleName
    FROM `User` u
    JOIN `Role` r
        ON u.RoleId = r.RoleId
    WHERE u.Email = p_email
      AND u.PasswordHash = p_password;
END $$
DELIMITER ;

DELIMITER //

CREATE PROCEDURE get_all_employees()
BEGIN
    SELECT
        u.UserId,
        u.FirstName,
        u.LastName,
        u.Email,
        u.PhoneNumber,
        u.RoleId,
        r.RoleName,
        u.DepartmentId,
        d.Name AS DepartmentName,
        u.DesignationtId,
        des.Name AS DesignationName,
        u.DateOfJoining,
        u.Status
    FROM `User` u
    LEFT JOIN `Role` r
        ON u.RoleId = r.RoleId
    LEFT JOIN `Departments` d
        ON u.DepartmentId = d.DepartmentId
    LEFT JOIN `Designations` des
        ON u.DesignationtId = des.DesignationId
    ORDER BY u.UserId DESC;
END //

DELIMITER ;



<--------------AttendanceReport------------>
DELIMITER //

CREATE PROCEDURE get_all_attendance()
BEGIN
    SELECT
        a.AttendanceId,
        a.UserId,
        u.FirstName,
        u.LastName,
        u.Email,
        a.Date,
        a.CheckIn,
        a.CheckOut,
        a.LunchIn,
        a.LunchOut,
        a.WorkingHours,
        a.ProductionHours,
        a.OvertimeHours,
        a.BreakHours,
        a.Late,
        a.Status
    FROM Attendance a
    LEFT JOIN `User` u
        ON a.UserId = u.UserId
    ORDER BY a.Date DESC, a.AttendanceId DESC;
END //

DELIMITER ;

DELIMITER $$
CREATE PROCEDURE sp_get_all_departments()
BEGIN
    SELECT
        DepartmentId,
        Name,
        NoOfEmployee,
        status,
        CreatedAt,
        CreatedBy,
        ModifiedBy,
        ModifiedAt

    FROM departments

    ORDER BY DepartmentId DESC;

END $$

DELIMITER ;



DELIMITER $$
CREATE PROCEDURE sp_add_department(
    IN p_name VARCHAR(255),
    IN p_status VARCHAR(50),
    IN p_created_by VARCHAR(255)
)
BEGIN
    INSERT INTO departments
    (
        Name,
        NoOfEmployee,
        status,
        CreatedAt,
        CreatedBy
    )

    VALUES
    (
        p_name,
        0,
        p_status,
        NOW(),
        p_created_by
    );

END $$

DELIMITER ;



-- UPDATE DEPARTMENT

DELIMITER $$
CREATE PROCEDURE sp_update_department(
    IN p_department_id INT,
    IN p_name VARCHAR(255),
    IN p_status VARCHAR(50),
    IN p_modified_by VARCHAR(255)
)
BEGIN

    UPDATE departments

    SET
        Name = p_name,
        status = p_status,
        ModifiedBy = p_modified_by,
        ModifiedAt = NOW()

    WHERE DepartmentId = p_department_id;

END $$

DELIMITER ;


-- DELETE DEPARTMENT

DELIMITER $$
CREATE PROCEDURE sp_delete_department(
    IN p_department_id INT
)
BEGIN

    DELETE FROM departments

    WHERE DepartmentId = p_department_id;

END $$

DELIMITER ;
