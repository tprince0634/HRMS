DELIMITER $$
CREATE PROCEDURE sp_login_user(
    IN p_email VARCHAR(255),
    IN p_password VARCHAR(255)
)

BEGIN
    SELECT
        u.UserId,
        u.Email,
        u.FirstName,
        u.LastName,
        u.RoleId,
        r.RoleName
    FROM `User` u
    JOIN `Role` r
        ON u.RoleId = r.RoleId
    WHERE u.Email = p_email
      AND u.PasswordHash = p_password;
END $$
DELIMITER ;

DELIMITER //

CREATE PROCEDURE get_all_employees()
BEGIN
    SELECT
        u.UserId,
        u.FirstName,
        u.LastName,
        u.Email,
        u.PhoneNumber,
        u.RoleId,
        r.RoleName,
        u.DepartmentId,
        d.Name AS DepartmentName,
        u.DesignationtId,
        des.Name AS DesignationName,
        u.DateOfJoining,
        u.Status
    FROM `User` u
    LEFT JOIN `Role` r
        ON u.RoleId = r.RoleId
    LEFT JOIN `Departments` d
        ON u.DepartmentId = d.DepartmentId
    LEFT JOIN `Designations` des
        ON u.DesignationtId = des.DesignationId
    ORDER BY u.UserId DESC;
END //

-- =========================================================
-- 6. SEARCH TRAINER BY NAME OR EMAIL
-- =========================================================

CREATE PROCEDURE search_trainer_by_name_or_email(
    IN p_name VARCHAR(255),
    IN p_email VARCHAR(255)
)
BEGIN

    SELECT
        TrainerId,
        FirstName,
        LastName,
        Role,
        Email,
        Description,
        Status,
        Phone,
        ProfilePicture
    FROM Trainer
    WHERE
        (
            p_name IS NOT NULL
            AND p_name <> ''
            AND (
                FirstName LIKE CONCAT('%', p_name, '%')
                OR LastName LIKE CONCAT('%', p_name, '%')
            )
        )
        OR
        (
            p_email IS NOT NULL
            AND p_email <> ''
            AND Email LIKE CONCAT('%', p_email, '%')
        )
    ORDER BY TrainerId DESC;

END $$

-- =========================================================
-- 9. GET TRAININGS BY TIME PERIOD
-- DAO: getTrainingsByTime(int page, int pageSize, String period)
--
-- Supported periods:
-- TODAY
-- LAST_7_DAYS
-- LAST_MONTH
-- =========================================================

CREATE PROCEDURE get_trainings_by_time(
    IN p_page INT,
    IN p_pageSize INT,
    IN p_period VARCHAR(50)
)
BEGIN

    DECLARE v_offset INT;

    SET v_offset = (p_page - 1) * p_pageSize;

    IF UPPER(p_period) = 'TODAY' THEN

        SELECT
            TrainingId,
            TrainerId,
            TrainingTypeId,
            UserId,
            TrainingCost,
            Description,
            Status,
            StartDate,
            EndDate,
            CreatedAt,
            CreatedBy,
            ModifiedBy,
            ModifiedAt
        FROM Training
        WHERE StartDate >= CURDATE()
          AND StartDate < CURDATE() + INTERVAL 1 DAY
        ORDER BY StartDate DESC
        LIMIT v_offset, p_pageSize;


    ELSEIF UPPER(p_period) = 'LAST_7_DAYS' THEN

        SELECT
            TrainingId,
            TrainerId,
            TrainingTypeId,
            UserId,
            TrainingCost,
            Description,
            Status,
            StartDate,
            EndDate,
            CreatedAt,
            CreatedBy,
            ModifiedBy,
            ModifiedAt
        FROM Training
        WHERE StartDate >= NOW() - INTERVAL 7 DAY
          AND StartDate <= NOW()
        ORDER BY StartDate DESC
        LIMIT v_offset, p_pageSize;


    ELSEIF UPPER(p_period) = 'LAST_MONTH' THEN

        SELECT
            TrainingId,
            TrainerId,
            TrainingTypeId,
            UserId,
            TrainingCost,
            Description,
            Status,
            StartDate,
            EndDate,
            CreatedAt,
            CreatedBy,
            ModifiedBy,
            ModifiedAt
        FROM Training
        WHERE StartDate >= NOW() - INTERVAL 1 MONTH
          AND StartDate <= NOW()
        ORDER BY StartDate DESC
        LIMIT v_offset, p_pageSize;

    ELSE

        SELECT
            TrainingId,
            TrainerId,
            TrainingTypeId,
            UserId,
            TrainingCost,
            Description,
            Status,
            StartDate,
            EndDate,
            CreatedAt,
            CreatedBy,
            ModifiedBy,
            ModifiedAt
        FROM Training
        ORDER BY StartDate DESC
        LIMIT v_offset, p_pageSize;

    END IF;

END $$

-- =========================================================
-- 7. GET TRAINING TYPES BY PAGE
-- DAO: getTrainingTypesByPage(
--      int page,
--      int pageSize,
--      String sortDirection
-- )
-- =========================================================

CREATE PROCEDURE get_training_types_by_page(
    IN p_page INT,
    IN p_pageSize INT,
    IN p_sortDirection VARCHAR(10)
)
BEGIN

    DECLARE v_offset INT;

    SET v_offset = (p_page - 1) * p_pageSize;

    IF UPPER(p_sortDirection) = 'DESC' THEN

        SELECT
            TrainingTypeId,
            TrainingTypeName,
            Description,
            Status
        FROM TrainingType
        ORDER BY TrainingTypeId DESC
        LIMIT v_offset, p_pageSize;

    ELSE

        SELECT
            TrainingTypeId,
            TrainingTypeName,
            Description,
            Status
        FROM TrainingType
        ORDER BY TrainingTypeId ASC
        LIMIT v_offset, p_pageSize;

    END IF;

END $$

-- 8. GET TRAINING TYPES BY TIME
-- DAO: getTrainingTypesByTime(
--      int page,
--      int pageSize,
--      String period
-- )
--
-- NOTE:
-- TrainingType table does NOT have CreatedAt/ModifiedAt/date
-- columns.
--
-- Therefore, actual time-based filtering is not possible
-- with the current table structure.
--
-- This procedure returns paginated records according to
-- the requested period parameter, but does NOT filter by
-- creation time.
-- =========================================================

CREATE PROCEDURE get_training_types_by_time(
    IN p_page INT,
    IN p_pageSize INT,
    IN p_period VARCHAR(50)
)
BEGIN

    DECLARE v_offset INT;

    SET v_offset = (p_page - 1) * p_pageSize;

    SELECT
        TrainingTypeId,
        TrainingTypeName,
        Description,
        Status
    FROM TrainingType
    ORDER BY TrainingTypeId DESC
    LIMIT v_offset, p_pageSize;

END $$

        ON u.RoleId = r.RoleId
    ORDER BY u.UserId DESC;
END$$
